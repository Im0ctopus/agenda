CREATE TABLE "Users" (
    Id SERIAL PRIMARY KEY,
    Email VARCHAR(255) NOT NULL
);



CREATE TABLE "Items" (
    Id SERIAL PRIMARY KEY,
    UserId INT NOT NULL,
    Name VARCHAR(255) NOT NULL,
    YEAR INT NOT NULL,
    MONTH INT NOT NULL,
    DAY INT NOT NULL,
    TIME TIME NOT NULL,
    Amount NUMERIC(10, 2) NOT NULL,
    Comment TEXT,  -- Optional comment for the item
    Done BOOLEAN NOT NULL DEFAULT FALSE,
    Paid BOOLEAN NOT NULL DEFAULT FALSE,
    FOREIGN KEY (UserId) REFERENCES "Users"(Id)  -- Establishes a foreign key relationship with the User table
);
